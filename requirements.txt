# Direct dependencies for gym-continuousDoubleAuction.
#
# This file lists only what the project imports directly. For the fully-resolved
# set (every transitive package, exact versions), see requirements-lock.txt.
#
# Install:
#     pip install -r requirements.txt
#     pip install -e .
#
# Python 3.12 (the only version CI tests; developed on 3.12). Ray 2.56
# supports a wider 3.10-3.14 range in principle, but this repo does not
# verify it - see setup.py's python_requires.

# --- RL stack -----------------------------------------------------------------
# Ray 2.56.x hard-pins gymnasium==1.2.2. Do NOT bump gymnasium independently:
# a newer gymnasium (1.3.x) is incompatible with this Ray release and pip will
# either refuse to resolve or silently break RLlib's env checks.
ray[rllib]==2.56.1
gymnasium==1.2.2

# torch is deliberately unpinned to a local (+cpu / +cu***) build so the same
# requirements file works on the CPU dev box and the GPU training machine.
# Select the right wheel index at install time - see requirements-lock.txt.
torch>=2.13.0,<3

# --- Numerics / data ----------------------------------------------------------
# Floor is 2.2, not 2.5: numpy dropped Python 3.10 wheels at 2.3.0 and Python
# 3.11 wheels at 2.5.0. CI only tests 3.12 now (see the note above), where
# >=2.5 would resolve fine either way, but this file is installed directly by
# `pip install -r requirements.txt` without going through setup.py's
# python_requires gate - so a >=2.5 floor would still silently break anyone
# who runs it on 3.10/3.11 despite the declared support range.
numpy>=2.2,<3
pandas>=3.0,<4
# Imported directly by train/episode_record.py and train/probe/corpus.py.
# It arrives transitively with ray[rllib], which is exactly the reasoning that
# produced the six/scikit-learn bug (doc/15 S3-6) - so it is declared.
pyarrow>=15
# scipy is imported nowhere in the project. Kept because the [plot] extra
# declares it and some matplotlib workflows want it, but it is NOT a direct
# dependency and this file said it was.
scipy>=1.18,<2

# --- Env internals ------------------------------------------------------------
sortedcontainers>=2.4      # order book price-level trees
# six is no longer imported anywhere: the orderbook package's `six.moves.cStringIO`
# was replaced by `io.StringIO` in the 2026-09-18 pass (doc/17 §45). It still
# arrives transitively via python-dateutil, which is why it stays in the lock.
tabulate>=0.10             # env render()

# --- Plotting / metrics -------------------------------------------------------
matplotlib>=3.11,<4
# tensorboardX is imported nowhere either; this line used to claim it was "used
# directly". Pinned only because ray[rllib] writes TensorBoard output through it
# and an unpinned transitive is worth naming.
tensorboardX>=2.6.5

# --- Optional -----------------------------------------------------------------
# GPUtil is only used by Ray for GPU system monitoring; harmless to omit on CPU.
GPUtil>=1.4
